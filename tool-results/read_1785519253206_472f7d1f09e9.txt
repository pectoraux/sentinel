   761→async function fetchIdentitySummary() {
   762→  const [
   763→    organizationsByType,
   764→    organizationsByStatus,
   765→    totalMembers,
   766→    totalDevices,
   767→    trustedDevices,
   768→    verificationsByStatus,
   769→    verificationsByType,
   770→    trustTiers,
   771→    topTrustProfiles,
   772→    recentVerifications,
   773→    recentOrgs,
   774→  ] = await Promise.all([
   775→    db.organization.groupBy({ by: ["type"], _count: true }),
   776→    db.organization.groupBy({ by: ["status"], _count: true }),
   777→    db.organizationMember.count({ where: { status: "active" } }),
   778→    db.device.count(),
   779→    db.device.count({ where: { status: "trusted" } }),
   780→    db.identityVerification.groupBy({ by: ["status"], _count: true }),
   781→    db.identityVerification.groupBy({ by: ["type"], _count: true }),
   782→    db.trustProfile.groupBy({ by: ["tier"], _count: true }),
   783→    db.trustProfile.findMany({
   784→      take: 5,
   785→      orderBy: { score: "desc" },
   786→      include: { user: { select: { id: true, email: true, name: true, image: true } } },
   787→    }),
   788→    db.identityVerification.findMany({
   789→      take: 8,
   790→      orderBy: { submittedAt: "desc" },
   791→      include: { user: { select: { id: true, email: true, name: true } } },
   792→    }),
   793→    db.organization.findMany({
   794→      take: 8,
   795→      orderBy: { createdAt: "desc" },
   796→      include: { _count: { select: { members: true } } },
   797→    }),
   798→  ]);
   799→
   800→  return {
   801→    organizations: {
   802→      byType: organizationsByType.map((g) => ({ type: g.type, count: g._count })),
   803→      byStatus: organizationsByStatus.map((g) => ({ status: g.status, count: g._count })),
   804→      total: organizationsByType.reduce((s, g) => s + g._count, 0),
   805→    },
   806→    members: { total: totalMembers },
   807→    devices: { total: totalDevices, trusted: trustedDevices, untrusted: totalDevices - trustedDevices },
   808→    verifications: {
   809→      byStatus: verificationsByStatus.map((g) => ({ status: g.status, count: g._count })),
   810→      byType: verificationsByType.map((g) => ({ type: g.type, count: g._count })),
   811→      total: verificationsByStatus.reduce((s, g) => s + g._count, 0),
   812→    },
   813→    trust: {
   814→      byTier: trustTiers.map((g) => ({ tier: g.tier, count: g._count })),
   815→      topProfiles: topTrustProfiles.map((p) => ({
   816→        userId: p.userId,
   817→        score: p.score,
   818→        tier: p.tier,
   819→        badges: p.badges ? JSON.parse(p.badges) : [],
   820→        user: p.user,
   821→      })),
   822→    },
   823→    recent: {
   824→      verifications: recentVerifications.map((v) => ({
   825→        id: v.id,
   826→        type: v.type,
   827→        status: v.status,
   828→        submittedAt: v.submittedAt,
   829→        user: v.user,
   830→      })),
   831→      organizations: recentOrgs.map((o) => ({
   832→        id: o.id,
   833→        key: o.key,
   834→        name: o.name,
   835→        type: o.type,
   836→        status: o.status,
   837→        country: o.country,
   838→        memberCount: o._count.members,
   839→        createdAt: o.createdAt,
   840→      })),
   841→    },
   842→  };
   843→}
