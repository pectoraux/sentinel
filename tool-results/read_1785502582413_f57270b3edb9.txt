   647→async function fetchIdentitySummary() {
   648→  const [
   649→    organizationsByType,
   650→    organizationsByStatus,
   651→    totalMembers,
   652→    totalDevices,
   653→    trustedDevices,
   654→    verificationsByStatus,
   655→    verificationsByType,
   656→    trustTiers,
   657→    topTrustProfiles,
   658→    recentVerifications,
   659→    recentOrgs,
   660→  ] = await Promise.all([
   661→    db.organization.groupBy({ by: ["type"], _count: true }),
   662→    db.organization.groupBy({ by: ["status"], _count: true }),
   663→    db.organizationMember.count({ where: { status: "active" } }),
   664→    db.device.count(),
   665→    db.device.count({ where: { status: "trusted" } }),
   666→    db.identityVerification.groupBy({ by: ["status"], _count: true }),
   667→    db.identityVerification.groupBy({ by: ["type"], _count: true }),
   668→    db.trustProfile.groupBy({ by: ["tier"], _count: true }),
   669→    db.trustProfile.findMany({
   670→      take: 5,
   671→      orderBy: { score: "desc" },
   672→      include: { user: { select: { id: true, email: true, name: true, image: true } } },
   673→    }),
   674→    db.identityVerification.findMany({
   675→      take: 8,
   676→      orderBy: { submittedAt: "desc" },
   677→      include: { user: { select: { id: true, email: true, name: true } } },
   678→    }),
   679→    db.organization.findMany({
   680→      take: 8,
   681→      orderBy: { createdAt: "desc" },
   682→      include: { _count: { select: { members: true } } },
   683→    }),
   684→  ]);
   685→
   686→  return {
   687→    organizations: {
   688→      byType: organizationsByType.map((g) => ({ type: g.type, count: g._count })),
   689→      byStatus: organizationsByStatus.map((g) => ({ status: g.status, count: g._count })),
   690→      total: organizationsByType.reduce((s, g) => s + g._count, 0),
   691→    },
   692→    members: { total: totalMembers },
   693→    devices: { total: totalDevices, trusted: trustedDevices, untrusted: totalDevices - trustedDevices },
   694→    verifications: {
   695→      byStatus: verificationsByStatus.map((g) => ({ status: g.status, count: g._count })),
   696→      byType: verificationsByType.map((g) => ({ type: g.type, count: g._count })),
   697→      total: verificationsByStatus.reduce((s, g) => s + g._count, 0),
   698→    },
   699→    trust: {
   700→      byTier: trustTiers.map((g) => ({ tier: g.tier, count: g._count })),
   701→      topProfiles: topTrustProfiles.map((p) => ({
   702→        userId: p.userId,
   703→        score: p.score,
   704→        tier: p.tier,
   705→        badges: p.badges ? JSON.parse(p.badges) : [],
   706→        user: p.user,
   707→      })),
   708→    },
   709→    recent: {
   710→      verifications: recentVerifications.map((v) => ({
   711→        id: v.id,
   712→        type: v.type,
   713→        status: v.status,
   714→        submittedAt: v.submittedAt,
   715→        user: v.user,
   716→      })),
   717→      organizations: recentOrgs.map((o) => ({
   718→        id: o.id,
   719→        key: o.key,
   720→        name: o.name,
   721→        type: o.type,
   722→        status: o.status,
   723→        country: o.country,
   724→        memberCount: o._count.members,
   725→        createdAt: o.createdAt,
   726→      })),
   727→    },
   728→  };
   729→}
