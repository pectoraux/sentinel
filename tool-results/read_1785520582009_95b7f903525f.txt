     2→// Sentinel — Prisma Schema (Development: SQLite)
     3→// =============================================================================
     4→// This schema runs on SQLite for the local sandbox so the platform boots
     5→// without a database server. It is structurally identical to the production
     6→// PostgreSQL schema (see schema.postgres.prisma).
     7→//
     8→// PostGIS geometry columns are intentionally NOT modeled in Prisma (Prisma has
     9→// no native geometry type). In production they are managed via raw SQL
    10→// migrations (see prisma/sql/postgis.sql) and accessed through repository
    11→// raw queries. This keeps the ORM models portable across both databases.
    12→// =============================================================================

   235→// =============================================================================
   236→// Event-Driven Architecture — Transactional Outbox
   237→// =============================================================================
   238→// Domain events are written to the outbox in the same DB transaction as the
   239→// state change that produced them, then a background relay publishes them to
   240→// the event bus. This guarantees at-least-once delivery without 2PC.
   241→// =============================================================================
   242→
   243→model OutboxEvent {
   244→  id            String   @id @default(cuid())
   245→  aggregateType String   // e.g. "User", "FeatureFlag"
   246→  aggregateId   String
   247→  eventType     String   // e.g. "user.created"
   248→  payload       String   // JSON string
   249→  metadata      String?  // JSON string (traceId, userId, etc.)
   250→  occurredAt    DateTime @default(now())
   251→  // status: pending | published | failed | dead_letter
   252→  status        String   @default("pending")
   253→  attempts      Int      @default(0)
   254→  lastError     String?
   255→  publishedAt   DateTime?
   256→  createdAt     DateTime @default(now())
   257→
   258→  @@index([status, occurredAt])
   259→  @@index([aggregateType, aggregateId])
   260→  @@index([eventType])
   261→}

   532→// MILESTONE 3 — Geospatial Platform (GIS Engine)
   533→// =============================================================================
   534→// PointOfInterest, SpatialRegion (polygons), GeoLayer (map layers).
   535→//
   536→// On SQLite (dev): lat/lng stored as Float, geojson as String (JSON).
   537→// On PostgreSQL+PostGIS (prod): numeric columns + native geography(Point/Polygon)
   538→// columns added via prisma/sql/postgis.sql (Prisma has no native geometry type).
   539→// The SpatialQueryService abstracts both: dev uses Haversine + ray-casting in
   540→// TypeScript; prod delegates to ST_DWithin / ST_Contains / ST_Distance.
   541→// =============================================================================
   542→
   543→model PointOfInterest {
   544→  id          String   @id @default(cuid())
   545→  name        String
   546→  type        String   // mining_site | water_body | forest_reserve | settlement | sensor_station | incident | checkpoint
   547→  lat         Float
   548→  lng         Float
   549→  // geojson: full GeoJSON Feature as JSON string (includes properties, geometry)
   550→  geojson     String?
   551→  layerId     String?
   552→  country     String?  // ISO code e.g. "GH"
   553→  region      String?
   554→  status      String   @default("active") // active | monitored | closed | verified | false_positive
   555→  severity    String?  // low | medium | high | critical (for incidents)
   556→  metadata    String?  // JSON: { confidence, source, reportCount, ... }
   557→  createdAt   DateTime @default(now())
   558→  updatedAt   DateTime @updatedAt
   559→
   560→  layer       GeoLayer? @relation(fields: [layerId], references: [id], onDelete: SetNull)
   561→
   562→  @@index([type])
   563→  @@index([status])
   564→  @@index([country])
   565→  @@index([layerId])
   566→  // Composite index for bounding-box queries (lng range then lat range)
   567→  @@index([lng, lat])
   568→}

   596→model GeoLayer {
   597→  id          String   @id @default(cuid())
   598→  key         String   @unique // e.g. "osm-base", "satellite", "mining-sites", "forest-reserves"
   599→  name        String
   600→  type        String   // base | overlay | data | vector
   601→  source      String   // osm | sentinel | landsat | custom | internal
   602→  description String?

  1086→// MILESTONE 10 — Civil Trust Engine
  1087→// =============================================================================
  1088→// Production trust system replacing simple reputation. Tracks 8 factors:
  1089→//   1. Accuracy       — verified reports / total reports
  1090→//   2. Reliability    — consistency over time (report frequency + verification rate)
  1091→//   3. False reports  — count of false-positive submissions (penalty)
  1092→//   4. Evidence quality — aggregate from M9 evidence weights
  1093→//   5. Contribution quality — comment/event quality (corroboration support rate)
  1094→//   6. Community impact  — positive impact (resolutions, helpful shares)
  1095→//   7. Decay          — trust decays if inactive (half-life 90 days)
  1096→//   8. Fraud resistance — fraud flags reduce trust; clean history boosts it
  1097→// =============================================================================

  1110→  fraudResistance Float    @default(1.0)   // 1.0 = clean, decreases per flag
  1111→  fraudFlagCount  Int      @default(0)

  1113→  totalReports    Int      @default(0)
  1114→  verifiedReports Int      @default(0)
  1115→  totalEvidence   Int      @default(0)
  1116→  totalComments   Int      @default(0)
  1117→  totalShares     Int      @default(0)
  1118→  lastActivityAt  DateTime?
